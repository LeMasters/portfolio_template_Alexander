{{- /* Optional visitor statistics, chosen in the Analytics form (data/analytics.yaml): service: none | goatcounter | cloudflare code: GoatCounter: the code chosen at sign-up ("janedoe"), or its address Cloudflare: the site token, or the whole snippet copied from Cloudflare Students may paste a whole address or snippet; the code is picked out of it. Nothing is added when the choice is "none", the code is missing or unrecognised (the build log then shows a warning), or the site is being previewed locally (hugo server), so a student's own previews are not counted. Both services set no cookies. */ -}} {{- $a := site.Data.analytics | default dict -}} {{- $service := lower (string ($a.service | default "none")) -}} {{- $raw := trim (string ($a.code | default "")) " \n\t" -}} {{- if and hugo.IsProduction (ne $service "none") -}} {{- if eq $service "goatcounter" -}} {{- $code := "" -}} {{- $c := lower $raw -}} {{- with findRESubmatch `([a-z0-9][a-z0-9-]*)\.goatcounter\.com` $c 1 }}{{ $code = index (index . 0) 1 }} {{- else }}{{ if findRE `^[a-z0-9][a-z0-9-]*$` $c 1 }}{{ $code = $c }}{{ end }}{{ end -}} {{- if $code }} {{- else }}{{ warnf "Analytics: GoatCounter is chosen but the code %q was not recognised; no analytics added." $raw }}{{ end -}} {{- else if eq $service "cloudflare" -}} {{- $token := "" -}} {{- with findRESubmatch `token\\?"\s*:\s*\\?"([A-Za-z0-9]+)` $raw 1 }}{{ $token = index (index . 0) 1 }} {{- else }}{{ if findRE `^[A-Za-z0-9]{16,64}$` $raw 1 }}{{ $token = $raw }}{{ end }}{{ end -}} {{- if $token }} {{- else }}{{ warnf "Analytics: Cloudflare is chosen but no site token was found in %q; no analytics added." $raw }}{{ end -}} {{- else -}} {{- warnf "Analytics: unknown service %q; expected none, goatcounter or cloudflare." $service -}} {{- end -}} {{- end -}}
